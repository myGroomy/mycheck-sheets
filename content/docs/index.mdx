## Simulator Checklist Interaktif

Komponen ini mendemonstrasikan cara kerja aksi checklist di mycheck — centang item, skip dengan alasan, atau isi nilai bebas.

```jsx
function ChecklistSimulator() {
  const [items, setItems] = React.useState([
    { id: 1, title: "Cek suhu freezer", type: "checkbox", state: "pending", value: "" },
    { id: 2, title: "Jumlah porsi siap saji", type: "number", state: "pending", value: "" },
    { id: 3, title: "Bersihkan area dapur", type: "checkbox", state: "pending", value: "" },
  ]);

  function updateItem(id, updates) {
    setItems((prev) => prev.map((item) => (item.id === id ? { ...item, ...updates } : item)));
  }

  return (
    <div className="border rounded-lg p-4 space-y-3 bg-gray-50">
      <p className="text-sm font-semibold text-gray-700">Simulasi Item Checklist</p>
      {items.map((item) => (
        <div key={item.id} className="flex items-center gap-2 text-sm">
          <input
            type="checkbox"
            checked={item.state === "done"}
            onChange={(e) => updateItem(item.id, { state: e.target.checked ? "done" : "pending" })}
            className="w-4 h-4"
          />
          <span className={item.state === "done" ? "line-through text-gray-400" : ""}>
            {item.title}
          </span>
          {item.type === "number" && (
            <input
              type="number"
              placeholder="isi angka"
              value={item.value}
              onChange={(e) => updateItem(item.id, { value: e.target.value, state: e.target.value ? "done" : "pending" })}
              className="border rounded px-2 py-1 text-xs w-24"
            />
          )}
          {item.state === "done" && (
            <span className="text-green-600 text-xs">Selesai</span>
          )}
        </div>
      ))}
      <button
        onClick={() => setItems((prev) => prev.map((i) => ({ ...i, state: "pending", value: "" })))}
        className="text-xs text-blue-600 underline"
      >
        Reset semua
      </button>
    </div>
  );
}

<ChecklistSimulator />
```
